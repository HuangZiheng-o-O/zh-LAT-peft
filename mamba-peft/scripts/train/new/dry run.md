LAT_MODEL=/home/user/mzs_h/model/second-gla-1.3B-100B/gla-1.3B-100B \
LAT_PREC=bf16 \
HF_HUB_OFFLINE=1 HF_EVALUATE_OFFLINE=1 TRANSFORMERS_OFFLINE=1 \
HF_HOME=/home/user/mzs_h/data/hf_cache \
HF_DATASETS_CACHE="$HF_HOME" \
HF_EVALUATE_CACHE="$HF_HOME" \
HP_DATA=sst2 EVAL_GEN=0 HP_VAL_SPLIT=test \

SWANLAB_MODE=cloud \
SWANLAB_PROJECT="delta_net-sst2-1-4090-Jan13" \
SWANLAB_EMAIL_ON_START=0 \
SWANLAB_EMAIL_ON_FINISH=0 \
SWANLAB_EMAIL_ON_INTERRUPT=0 \
./lat_batch_tmux.sh \
  --suite E142 \
  --round all \
  --pairs "87:glue-tvt_sst2" \
  --gpus "0" \
  --gpu-plan "1" \
